def check_same(x, y, n, matrix):
    # N이 1인 경우 자신의 값을 문자열로 Return
    if n == 1:
        return str(matrix[x][y])    first_val = matrix[x][y]
    
    for i in range(x, x + n):
        for j in range(y, y + n):
            if matrix[i][j] != first_val:
                half = n // 2
                # 4등분하여 각각 check_same 호출
                top_left = check_same(x, y, half, matrix)
                top_right = check_same(x, y + half, half, matrix)
                bottom_left = check_same(x + half, y, half, matrix)
                bottom_right = check_same(x + half, y + half, half, matrix)
                return f"({top_left}{top_right}{bottom_left}{bottom_right})"
                
    return str(first_val)

N = int(input())

matrix = [list(map(int, input().strip())) for _ in range(N)]

print(N)
for row in matrix:
    print("".join(map(str, row)))

compress(N, matrix)
